{{ t("A dataset is an append-friendly ledger. Here you only decide the shape of the columns; rows come from ingest.")|safe }}

name
{{ t("Dataset name (used in the URL and directory name).") }}
key
{{ t("Column that makes a row unique. Re-appending the same key upserts (last write wins).")|safe }}
kind
{{ t("Display label for the kind (tabular / detection / sfm / llm …). Optional.") }}
columns[].type
str / int / float / bool / datetime, {{ t("lists are") }} list[float] / list[int] / list[str], {{ t("and undecided nesting is") }} json.
columns[].required
{{ t("true makes it mandatory (rows missing it are rejected).") }}
nested
{{ t("A structured child table such as a time series. Kept inline in the row JSON, hidden from the list view, and read at eval time with")|safe }} ctx.dataset.frames(id) {{ t("(this is where structured ground truth goes).") }}
{{ t("Heavy assets") }}
{{ t("Raw waveforms, images, masks and point clouds do not go in nested. Put a URL/path in a scalar column and read it at eval time with")|safe }} ctx.read_bytes(url) {{ t("(the ledger holds only the reference).") }}

{{ t("Example") }}

name: sensor-scans
kind: sensor
key: sample_id
columns:
  - {name: sample_id,    type: str, required: true}
  - {name: site,         type: str}
  - {name: calibrated,   type: bool}
  - {name: waveform_url, type: str}  # heavy assets: reference only, read at eval time
  - {name: gt,           type: int, required: true}
nested:
  readings:                          # the raw series. inline in the row, hidden from the list
    schema:
      - {name: t,     type: float}
      - {name: value, type: float}

{{ t("Writing table.yaml") }}

{{ t("An eval table is the list where runs stack up as stones. Here you decide the columns shown in that list.")|safe }}

dataset
{{ t("Default target. Filled in by the selector on the left (you may also write it in the YAML; if the selector is empty the YAML wins).")|safe }}
columns[].primary
{{ t("Primary metric = stone width. Put")|safe }} primary: true {{ t("on exactly one column (defaults to the first).")|safe }}
columns[].direction
{{ t("Which way is better.") }} higher {{ t("(bigger is better) /") }} lower {{ t("(smaller is better, e.g. a miss rate). Decides delta colors and stone direction.") }}
columns[].scale
{{ t("Absolute range for the stone width,") }} [min, max]. {{ t("With it, small differences are not exaggerated. Without it, the scale is relative to the runs on screen.") }}
columns[].display
{{ t("Header label in the list view. Defaults to name.") }}
default_sort
{{ t("Default ordering (e.g.") }} created_at desc).

{{ t("Columns must match exactly the row your Evaluator returns (common metadata such as eval_id / snapshot_id / evaluator_version is added by cairn).")|safe }}

{{ t("Example") }}

name: scan-anomaly
columns:
  - {name: f1,        type: float, display: "F1",
     primary: true, direction: higher, scale: [0, 1]}
  - {name: miss_rate, type: float, display: "missed%",
     direction: lower}
  - {name: coverage,  type: float, display: "coverage%"}
default_sort: created_at desc
